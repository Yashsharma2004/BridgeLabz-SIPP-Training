CLASS Pet
    ATTRIBUTES: name, age
    CONSTRUCTOR(name, age):
        SET this.name = name
        SET this.age = age
    METHOD displayInfo()

CLASS Dog EXTENDS Pet
    ATTRIBUTE: breed
    CONSTRUCTOR(name, age, breed):
        CALL super(name, age)
        SET this.breed = breed
    METHOD displayInfo()

CLASS Cat EXTENDS Pet
    ATTRIBUTE: isIndoor
    CONSTRUCTOR(name, age, isIndoor):
        CALL super(name, age)
        SET this.isIndoor = isIndoor
    METHOD displayInfo()

CLASS Bird EXTENDS Pet
    ATTRIBUTE: color
    CONSTRUCTOR(name, age, color):
        CALL super(name, age)
        SET this.color = color
    METHOD displayInfo()

MAIN
    CREATE Dog d = NEW Dog("Bruno", 3, "Labrador")
    CREATE Cat c = NEW Cat("Milo", 2, true)
    CREATE Bird b = NEW Bird("Rio", 1, "Blue")
    CALL d.displayInfo()
    CALL c.displayInfo()
    CALL b.displayInfo()
