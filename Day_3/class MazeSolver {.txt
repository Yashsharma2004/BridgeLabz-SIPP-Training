class MazeSolver {
    int[][] maze;
    boolean[][] visited;
    int n;

    MazeSolver(int[][] maze) {
        this.maze = maze;
        n = maze.length;
        visited = new boolean[n][n];
    }

    boolean solve(int x, int y) {
        if(x == n-1 && y == n-1) return true; // reached end
        if(x<0 || y<0 || x>=n || y>=n || maze[x][y]==0 || visited[x][y]) return false;

        visited[x][y] = true;
        if(solve(x+1,y) || solve(x,y+1) || solve(x-1,y) || solve(x,y-1)) return true;
        visited[x][y] = false; // backtrack
        return false;
    }
}
